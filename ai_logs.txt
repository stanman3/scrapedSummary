from urllib.parse import urlparse

import requests
from bs4 import BeautifulSoup, Tag

TIMEOUT_SECONDS = 10
HEADERS = {"User-Agent": "Mozilla/5.0 (compatible; ArticleSummarizer/0.1)"}
ALLOWED_SCHEMES = {"http", "https"}

NOISE_TAGS = ["script", "style", "nav", "footer", "aside", "form"]
MIN_PARAGRAPH_CHARS = 40
MIN_ARTICLE_CHARS = 200


class ScraperError(Exception):
    """Raised when the article can't be downloaded or parsed."""


def fetch_html(url: str) -> str:
    if urlparse(url).scheme not in ALLOWED_SCHEMES:
        raise ScraperError("Only http and https URLs are supported")

    try:
        resp = requests.get(url, headers=HEADERS, timeout=TIMEOUT_SECONDS)
        resp.raise_for_status()
    except requests.RequestException as exc:
        raise ScraperError(f"Could not fetch the page: {exc}") from exc

    # Without a charset in the headers, requests assumes ISO-8859-1,
    # which garbles Cyrillic and symbols like €. Fall back to detection.
    if resp.encoding is None or resp.encoding.lower() == "iso-8859-1":
        resp.encoding = resp.apparent_encoding

    return resp.text


def extract_article(html: str) -> tuple[str, str]:
    """Return (title, text) of the article, or raise ScraperError."""
    soup = BeautifulSoup(html, "html.parser")

    title = _extract_title(soup)

    for tag in soup(NOISE_TAGS):
        tag.decompose()

    text = ""
    article = soup.find("article")
    if article:
        text = _paragraphs_text(article)

    # No <article>, or it was (almost) empty: use all paragraphs on the page.
    if len(text) < MIN_ARTICLE_CHARS:
        text = _paragraphs_text(soup.body or soup)

    if len(text) < MIN_ARTICLE_CHARS:
        raise ScraperError(
            "Could not extract article text (page may be paywalled or JS-rendered)"
        )

    return title, text


def _extract_title(soup: BeautifulSoup) -> str:
    h1 = soup.find("h1")
    if h1 and h1.get_text(strip=True):
        return _clean(h1.get_text(" "))
    if soup.title and soup.title.get_text(strip=True):
        return _clean(soup.title.get_text(" "))
    return ""


def _paragraphs_text(container: Tag) -> str:
    paragraphs = []
    for p in container.find_all("p"):
        text = _clean(p.get_text(" "))
        if len(text) >= MIN_PARAGRAPH_CHARS:
            paragraphs.append(text)
    return "\n\n".join(paragraphs)


def _clean(text: str) -> str:
    return " ".join(text.split())

Review
What’s good

The scheme check runs before any network call, and requests errors are wrapped in your own ScraperError, so main.py only needs to catch one exception type.
The requests ISO-8859-1 pitfall is handled, and the comment explains why.
It has a sensible fallback chain: <article>, then all <p> tags, then an error. The tuning constants sit at the top of the file.
fetch_html and extract_article are separate, so parsing is testable without the network.
The title is read before the noise tags are removed.
Issues, most important first

form in NOISE_TAGS can wipe whole pages. Older ASP.NET-style sites wrap the entire body in one <form>, and decompose() would delete all the content. You’d get a misleading “paywalled or JS-rendered” error. Either drop form from the list, or remove only forms that don’t contain the article.
SSRF. A user can submit http://localhost:... or http://169.254.169.254/..., and your server will fetch it. You don’t need to fix this for a prototype, but put it in the README as a known limitation with the fix (resolve the host and block private IP ranges). It’s a good interview talking point.
No check on the response type or size. A link to a PDF or a multi-GB file is downloaded in full. A cheap fix is to check that Content-Type contains html, and to use stream=True with a byte cap.
The <h1> may be the site logo. Some sites put the brand in an <h1>. Preferring the <h1> inside <article>, or og:title, is more robust. This is a minor point.
Paragraphs under 40 characters are dropped. That’s a reasonable trade-off, but mention it in the README.
Also cap the text length before sending it to the LLM. That belongs in summarizer.py, not here.

from unittest.mock import Mock, patch

import pytest
import requests

from scraper import ScraperError, extract_article, fetch_html

# Three sample paragraphs, each longer than MIN_PARAGRAPH_CHARS (40) and
# together longer than MIN_ARTICLE_CHARS (200), so they count as an article.
PARA_1 = "The central bank held interest rates steady on Thursday, citing persistent inflation in the services sector."
PARA_2 = "Analysts at several banks had expected a cut, and bond yields rose sharply after the announcement."
PARA_3 = "Markets in Europe closed lower, with financial stocks leading the decline across most major exchanges."


def _page(body: str, head: str = "") -> str:
    return f"<html><head>{head}</head><body>{body}</body></html>"


# ---------- extract_article ----------


def test_extracts_title_and_article_paragraphs():
    html = _page(
        f"""
        <nav><p>Home | Markets | Economy | Sign in to your account today</p></nav>
        <h1>  Central bank   holds rates </h1>
        <article>
            <p>{PARA_1}</p>
            <p>{PARA_2}</p>
            <p>{PARA_3}</p>
            <aside><p>Subscribe to our premium newsletter for exclusive content.</p></aside>
            <script>var tracking = "this paragraph should never appear in output";</script>
        </article>
        <footer><p>Copyright 2026 Example Media. All rights reserved worldwide.</p></footer>
        """
    )

    title, text = extract_article(html)

    assert title == "Central bank holds rates"  # whitespace normalised
    assert text == f"{PARA_1}\n\n{PARA_2}\n\n{PARA_3}"
    assert "Subscribe" not in text
    assert "Copyright" not in text
    assert "tracking" not in text


def test_falls_back_to_all_paragraphs_when_no_article_tag():
    html = _page(
        f"<div><p>{PARA_1}</p><p>too short</p><p>{PARA_2}</p><p>{PARA_3}</p></div>"
    )

    _, text = extract_article(html)

    assert text == f"{PARA_1}\n\n{PARA_2}\n\n{PARA_3}"  # "too short" is dropped


def test_title_falls_back_to_title_tag_then_empty():
    body = f"<p>{PARA_1}</p><p>{PARA_2}</p><p>{PARA_3}</p>"

    title, _ = extract_article(_page(body, head="<title>Markets Wrap</title>"))
    assert title == "Markets Wrap"

    title, _ = extract_article(_page(body))
    assert title == ""


def test_raises_when_there_is_too_little_text():
    html = _page(f"<h1>Paywalled</h1><p>{PARA_1}</p>")  # ~100 chars < 200

    with pytest.raises(ScraperError):
        extract_article(html)

@pytest.mark.parametrize("url", ["ftp://example.com/a", "file:///etc/passwd", "example.com"])
def test_fetch_html_rejects_non_http_urls(url):
    with patch("scraper.requests.get") as mock_get:
        with pytest.raises(ScraperError):
            fetch_html(url)
    mock_get.assert_not_called()  # rejected before any network call


def test_fetch_html_returns_page_text():
    resp = Mock(text="<html>ok</html>", encoding="utf-8")
    with patch("scraper.requests.get", return_value=resp) as mock_get:
        assert fetch_html("https://example.com/a") == "<html>ok</html>"
    assert mock_get.call_args.kwargs["timeout"] == 10


def test_fetch_html_wraps_network_and_http_errors():
    with patch("scraper.requests.get", side_effect=requests.Timeout("slow")):
        with pytest.raises(ScraperError, match="Could not fetch"):
            fetch_html("https://example.com/a")

    resp = Mock()
    resp.raise_for_status.side_effect = requests.HTTPError("404")
    with patch("scraper.requests.get", return_value=resp):
        with pytest.raises(ScraperError):
            fetch_html("https://example.com/a")


def test_fetch_html_uses_detected_encoding_when_header_says_latin1():
